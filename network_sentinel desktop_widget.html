Código Python Real para Windows (PyQt6 / QSystemTray)

Ejecuta este script en Windows. Usa el API nativo de Windows (o `psutil`/`netifaces`) para escuchar cambios de Gateway en tiempo real y mostrar este widget flotante transparente: 

import sys, time, socket, urllib.request, subprocess
from PyQt6.QtCore import Qt, QTimer, QThread, pyqtSignal
from PyQt6.QtWidgets import (QApplication, QWidget, QVBoxLayout, QLabel, 
                             QHBoxLayout, QFrame, QPushButton, QSystemTrayIcon)

class NetworkMonitor(QWidget):
    def __init__(self):
        super().__init__()
        # Windows 11 Frameless & Always On Top
        self.setWindowFlags(Qt.WindowType.FramelessWindowHint | Qt.WindowType.WindowStaysOnTopHint | Qt.WindowType.SubWindow)
        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground)
        self.resize(340, 240)
        self.last_gateway = ""

        # Layout UI
        layout = QVBoxLayout(self)
        self.card = QFrame()
        self.card.setStyleSheet("""
            background-color: rgba(20, 26, 38, 220);
            border: 1px solid rgba(255, 255, 255, 0.15);
            border-radius: 18px;
            color: white;
            font-family: 'Segoe UI';
        """)
        card_layout = QVBoxLayout(self.card)

        self.title_lbl = QLabel("🛡️ DETECTOR DE RED ACTIVA")
        self.title_lbl.setStyleSheet("font-weight: bold; font-size: 11px; color: #38bdf8;")
        self.status_lbl = QLabel("Detectando...")
        self.status_lbl.setStyleSheet("font-size: 15px; font-weight: bold;")
        self.ip_lbl = QLabel("IP Local: -- | GW: --")
        self.ip_lbl.setStyleSheet("font-family: Consolas; font-size: 12px; color: #a5f3fc;")

        card_layout.addWidget(self.title_lbl)
        card_layout.addWidget(self.status_lbl)
        card_layout.addWidget(self.ip_lbl)
        layout.addWidget(self.card)

        # Polling Timer (Cada 2 segundos)
        self.timer = QTimer(self)
        self.timer.timeout.connect(self.check_network)
        self.timer.start(2000)
        self.check_network()

    def check_network(self):
        try:
            # Obtener Gateway predeterminado en Windows
            res = subprocess.check_output("route print 0.0.0.0", shell=True).decode()
            gw = [l.split()[2] for l in res.splitlines() if "0.0.0.0" in l and len(l.split()) > 4][0]
            # IP local
            s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
            s.connect(("8.8.8.8", 80))
            local_ip = s.getsockname()[0]
            s.close()

            # Clasificar Segmento
            env = "Wi-Fi Casa / Genérica"
            if gw.startswith("10.8.") or gw.startswith("10.0."):
                env = "🛡️ VPN Corporativa"
            elif gw.startswith("192.168.100."):
                env = "📹 VLAN Cámaras IoT"
            elif gw.startswith("192.168.43.") or gw.startswith("172.20.10."):
                env = "📱 Hotspot Celular"
            elif gw.startswith("172.16."):
                env = "🏢 Red de Empleados"

            self.status_lbl.setText(env)
            self.ip_lbl.setText(f"IP: {local_ip}\nGW: {gw}")

            # Notificar si cambió
            if self.last_gateway and self.last_gateway != gw:
                print(f"[CAMBIO DETECTADO] Salto de {self.last_gateway} a {gw}")
            self.last_gateway = gw
        except Exception as e:
            self.status_lbl.setText("Desconectado / Error")

if __name__ == "__main__":
    app = QApplication(sys.argv)
    w = NetworkMonitor()
    w.show()
    sys.exit(app.exec())
